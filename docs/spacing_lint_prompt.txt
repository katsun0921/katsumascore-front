あなたはフロントエンドアーキテクトです。
KatsumaScoreプロジェクトにおいて、Spacing Design Token + ESLint強制ルールを実装してください。

# ■ 前提
- Tailwindはlayoutとspacingの責務を持つ
- Design Tokenは視覚表現を担う
- SCSSでspacing定義は禁止

# ■ 目的
- spacingの完全統制
- 任意値・直書きの排除
- UI崩れ防止

# ■ 3. ESLintルール

ファイル:
eslint-rules/no-direct-spacing.mjs

実装:

export default {
  meta: {
    type: "problem",
    docs: {
      description: "Disallow direct spacing usage",
    },
  },
  create(context) {
    return {
      JSXAttribute(node) {
        if (node.name.name !== "style") return;

        const value = node.value?.expression;
        if (!value || value.type !== "ObjectExpression") return;

        value.properties.forEach((prop) => {
          const key = prop.key.name;

          if (["margin", "padding", "gap"].includes(key)) {
            context.report({
              node,
              message: "spacingの直接指定は禁止です。Tokenを使用してください。",
            });
          }
        });
      },

      Literal(node) {
        if (typeof node.value !== "string") return;

        if (node.value.match(/(p|m|gap)-\[(.*?)\]/)) {
          context.report({
            node,
            message: "任意値spacingは禁止です。",
          });
        }
      },
    };
  },
};

# ■ 4. ESLint設定

import noDirectSpacing from "./eslint-rules/no-direct-spacing.mjs";

export default [
  {
    plugins: {
      katsuma: {
        rules: {
          "no-direct-spacing": noDirectSpacing,
        },
      },
    },
    rules: {
      "katsuma/no-direct-spacing": "error",
    },
  },
];
